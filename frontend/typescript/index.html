ide dasar

JavaScript vs TypeScript

✗ JavaScript
function total(cart) {
  return cart.items.reduce(
    (sum, i) => sum + i.price * i.qty, 0);
}
total({ itms: [] });   // typo, error saat jalan
✓ TypeScript
type Item = { price: number; qty: number };
type Cart = { items: Item[] };

function total(cart: Cart): number {
  return cart.items.reduce(
    (sum, i) => sum + i.price * i.qty, 0);
}
total({ itms: [] });   // merah di editor!
Bonus gratis: autocomplete jadi pintar.

tipe
type Payment =
  | { kind: "card"; last4: string }
  | { kind: "va"; bank: string; number: string }
  | { kind: "qris" };
pemakaian
function label(p: Payment) {
  switch (p.kind) {
    case "card": return `Kartu •••• ${p.last4}`;
    case "va":   return `VA ${p.bank}`;
    case "qris": return "QRIS";
  }
}
// tambah kind baru → compiler menunjuk
// switch yang lupa diperbarui
Di dalam case "card", TypeScript tahu p.last4 pasti ada.
