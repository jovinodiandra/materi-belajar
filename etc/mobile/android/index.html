Tampilan dengan Jetpack Compose

✗ Cara lama: XML + findViewById
// layout.xml + Activity
val tv = findViewById<TextView>(R.id.count)
btn.setOnClickListener {
  count++
  tv.text = "$count" // ← update manual
}
✓ Compose: UI ikut state
@Composable
fun Counter() {
  var count by remember { mutableStateOf(0) }
  Button(onClick = { count++ }) {
    Text("Diklik $count kali")
  } // ← UI update otomatis
}

arsitektur

ViewModel + StateFlow + Compose

ViewModel
class CartViewModel(
  private val repo: CartRepository
) : ViewModel() {
  val state = repo.items()
    .map { CartUi(it, it.sumOf { i -> i.price }) }
    .stateIn(viewModelScope,
             SharingStarted.WhileSubscribed(5000),
             CartUi.Empty)
}
Composable
@Composable
fun CartScreen(vm: CartViewModel = hiltViewModel()) {
  val ui by vm.state.collectAsStateWithLifecycle()
  LazyColumn {
    items(ui.items, key = { it.id }) { ItemRow(it) }
  }
  Text("Total: ${ui.total}")
}
viewModelScope otomatis membatalkan kerja saat layar benar-benar ditutup.
