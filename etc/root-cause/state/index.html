    items = userItems;

  }

contoh

State di field singleton vs di dalam fungsi

✗ Field di bean singleton
@Service
class HitungService {
  private int total; // ← dibagi semua request!

  int hitung(List<Item> items) {
    total = 0;
    items.forEach(i -> total += i.harga);
    return total;
  }
}
✓ Variabel lokal
@Service
class HitungService {
  int hitung(List<Item> items) {
    int total = 0; // ← milik request ini saja
    for (Item i : items) total += i.harga;
    return total;
  }
}
Bean Spring itu singleton: satu objek dipakai semua request bersamaan.
kebiasaan baik

Aturan main mengelola state

  • Buat service stateless: simpan state di DB/Redis, bukan di memori server.
  • Hindari field yang bisa berubah di class singleton.
  • Batasi perpindahan status pakai state machine (PAID baru boleh ke SHIPPED).
  • Satu sumber kebenaran (single source of truth) untuk setiap data.
status

Jaga perpindahan status

✗ status bebas diubah
order.setStatus("SHIPPED");
// dari CANCELLED pun bisa jadi SHIPPED
// dari mana saja, kapan saja
✓ perpindahan yang sah saja
enum Status {
  NEW, PAID, SHIPPED, CANCELLED;
  boolean canMoveTo(Status next) {
    return switch (this) {
      case NEW  -> next == PAID || next == CANCELLED;
      case PAID -> next == SHIPPED || next == CANCELLED;
      default   -> false;
    };
  }
}
State yang tidak mungkin harus benar-benar tidak bisa terjadi di kode.
